> [3346. 执行操作后元素的最高频率 I](https://leetcode.cn/problems/maximum-frequency-of-an-element-after-performing-operations-i/description/)

![[Pasted image 20260305212029.png]]

```go
func maxFrequency(nums []int, k int, numOperations int) int {
    counter := make(map[int]int)
    diff := make(map[int]int)
    for _, num := range nums {
        counter[num]++
        diff[num] += 0
        diff[num - k]++
        diff[num + k + 1]--
    }

    var ans, sum int
    for _, num := range slices.Sorted(maps.Keys(diff)) {
        sum += diff[num]
        ans = max(ans, min(sum, counter[num] + numOperations))
    }
    return ans 
}
```
